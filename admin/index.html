-- 1. جدول ملفات الطلاب الشخصية (Profiles) create table if not exists public.profiles ( id uuid references auth.users on delete cascade primary key, full_name text, phone text, last_active_token text, created_at timestamp with time zone default timezone('utc'::text, now()) not null ); -- 2. جدول إعلانات المنصة (Announcements) create table if not exists public.announcements ( id uuid default gen_random_uuid() primary key, content text not null, created_at timestamp with time zone default timezone('utc'::text, now()) not null ); -- 3. جدول المذكرات والملفات (Notes) create table if not exists public.notes ( id uuid default gen_random_uuid() primary key, title text not null, file_url text not null, created_at timestamp with time zone default timezone('utc'::text, now()) not null ); -- 4. جدول بطل الفصل والنقاط (Champions) create table if not exists public.champions ( id uuid default gen_random_uuid() primary key, student_name text not null, points integer default 0 not null, student_rank integer, score text, created_at timestamp with time zone default timezone('utc'::text, now()) not null ); -- 5. جدول أسئلة المنتدى والردود المكتوبة والصوتية (Forum) create table if not exists public.forum ( id uuid default gen_random_uuid() primary key, user_id uuid references auth.users on delete cascade, user_name text, content text, admin_reply_text text, admin_reply_voice text, created_at timestamp with time zone default timezone('utc'::text, now()) not null ); -- 6. تفعيل صلاحيات القراءة والعمليات العامة على الجداول (RLS Policies) alter table public.profiles enable row level security; alter table public.announcements enable row level security; alter table public.notes enable row level security;
