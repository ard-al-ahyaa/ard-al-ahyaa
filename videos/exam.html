async function renderAssessments(index){ const lecture=currentList[index]; const homework=$("homework-"+index); const exam=$("exam-"+index); if(!lecture||!homework||!exam)return; const assessments=await getAssessments(lecture.id); if(!homework.isConnected)return; function cards(type){ const items=assessments.filter(x=>x.assessment_type===type); if(!items.length){ return `
لا توجد ${type==="homework"?"واجبات":"امتحانات"} متاحة حاليًا.
`; } return items.map(item=>`
${escapeHtml(item.title)}

${item.duration_minutes ?"المدة المحددة: "+escapeHtml(item.duration_minutes)+" دقيقة" :"اختبار إلكتروني"}

ابدأ ${type==="homework"?"الواجب":"الامتحان"}
`).join(""); } homework.innerHTML=cards("homework"); exam.innerHTML=cards("exam"); }
